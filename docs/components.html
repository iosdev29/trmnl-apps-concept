All components

Every TRMNL UI component on one screen, and the ones that come next.

TRMNL UI 1.0 · App API v1 · TRMNL X · TRMNL OG
gallery.ts
// Name: Components
import { App, View, Layout, Title, Description, Row, Value, ProgressBar, Divider, List, TitleBar, state, register } from "./trmnl-ui";
export { trmnl_event } from "./trmnl-ui";

const focus = state("focus", 1);

class Gallery extends App {
  body(): View {
    return Layout([
      Title("Components"),
      Description("Every TRMNL UI component on one screen, rendered by the firmware."),
      Row([Value("42", "Value"), Value("7°", "with a label"), Value("3/8", "in a Row")]),
      ProgressBar(5, 8),
      Divider(),
      List(["List row", "Focused row", "Another row"], focus),
      TitleBar("TitleBar", "instance label"),
    ]);
  }
}

register(new Gallery());

Title, Description, Row of Values, ProgressBar, Divider, List and TitleBar.
