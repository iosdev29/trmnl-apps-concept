glance()

The screen the playlist shows for your app.

TRMNL UI 1.0 · App API v1 · TRMNL X · TRMNL OG

Declaration

glance(): View | null

Return Value

A view, or null to show the current body().

Discussion

An installed app is a playlist item, like a plugin. When the playlist reaches it, the firmware calls the app with onLaunch(GLANCE) and shows this view. No press reaches the app in this mode: Select opens it, and Hold middle in the app returns to this same playlist item.

water.ts
// Name: Water
import { App, View, Layout, Title, Description, Value, ProgressBar, List, TitleBar, state, register } from "./trmnl-ui";
export { trmnl_event } from "./trmnl-ui";

const GOAL = 8;
const glasses = state("glasses");
const choice = state("choice");

function act(row: i32): void {
  if (row == 0 && glasses.value < GOAL) glasses.value = glasses.value + 1;
  if (row == 1) glasses.value = 0;
}

class Water extends App {
  body(): View {
    return Layout([
      Value(glasses.value.toString() + " / " + GOAL.toString(), "glasses today"),
      ProgressBar(glasses.value, GOAL),
      List(["Drank a glass", "Start a new day"], choice).onSelect(act),
      TitleBar("Water"),
    ]);
  }
  // What the playlist shows between plugins.
  glance(): View {
    return Layout([
      Title("Water"),
      Description(glasses.value.toString() + " of " + GOAL.toString() + " glasses today."),
      ProgressBar(glasses.value, GOAL),
      TitleBar("Water", "Select to log a glass"),
    ]);
  }
}

register(new Water());

The Water glance in the playlist.
